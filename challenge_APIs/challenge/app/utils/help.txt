class Logger:
    def __set_logger(self):
        log_directory = "utils/logs" # Ruta dónde se guardan los logs
        log_filename = "app.log"

        logger = logging.getLogger(__name__) # Obtenemos un logger con el nombre de la aplicación
        logger.setLevel(logging.DEBUG) # Nivel de Log

        log_path = os.path.join(log_directory, log_filename) # Unimos el directorio con el nombre del archivo
        file_handler = logging.FileHandler(log_path, encoding='utf-8') # Codificación del lenguaje
        file_handler.setLevel(logging.DEBUG)

        formatter = logging.Formatter('%(asctime)s - %(levelname)s - [%(name)s] - %(message)s', "%Y-%m-%d %H:%M%S") # Formato
        file_handler.setFormatter(formatter)

        if (logger.hasHandlers()):
            logger.handlers.clear()

        logger.addHandler(file_handler)

        return logger
    
    @classmethod
    def add_to_log(cls, level, message):
        try:
            logger = cls.__set_logger(cls)

            if (level == "critical"):
                logger.critical(message)
            elif (level == "debug"):
                logger.debug(message)
            elif (level == "error"):
                logger.error(message)
            elif (level == "info"):
                logger.info(message)
            elif (level == "warning"):
                logger.warm(message)
        except Exception as ex:
            print(traceback.format_exc()) # traceback is not define
            print(ex)
